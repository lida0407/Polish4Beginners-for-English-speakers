/** h('div', {class: 'card', onclick: fn}, 'text', node, [more], null) -> Element */
export function h(tag, attrs, ...kids) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs || {})) {
    if (v === null || v === undefined || v === false) continue;
    if (k === 'class') el.className = v;
    else if (k === 'style' && typeof v === 'object') Object.assign(el.style, v);
    else if (k === 'dataset') Object.assign(el.dataset, v);
    else if (k.startsWith('on') && typeof v === 'function') el.addEventListener(k.slice(2).toLowerCase(), v);
    else if (k === 'value' || k === 'checked' || k === 'disabled' || k === 'selected') el[k] = v;
    else el.setAttribute(k, v === true ? '' : v);
  }
  const add = (kid) => {
    if (kid === null || kid === undefined || kid === false) return;
    if (Array.isArray(kid)) kid.forEach(add);
    else el.append(kid instanceof Node ? kid : document.createTextNode(String(kid)));
  };
  kids.forEach(add);
  return el;
}

// Everything the learner owns lives in localStorage under "p4b:<key>" as JSON.
// Each key carries a last-changed time so sign-in sync can merge per key.

const PREFIX = 'p4b:';
const listeners = new Set();
let stamps = readRaw('_t', {});

function readRaw(key, def) {
  try {
    const raw = localStorage.getItem(PREFIX + key);
    return raw === null ? def : JSON.parse(raw);
  } catch (e) {
    return def;
  }
}

export const store = {
  get(key, def = null) {
    return readRaw(key, def);
  },
  set(key, value, { silent = false, at = Date.now() } = {}) {
    localStorage.setItem(PREFIX + key, JSON.stringify(value));
    stamps[key] = at;
    localStorage.setItem(PREFIX + '_t', JSON.stringify(stamps));
    if (!silent) listeners.forEach((fn) => fn(key, value));
  },
  /** store.update('fav', (list) => [...list, id], []) */
  update(key, fn, def = null) {
    const next = fn(this.get(key, def));
    this.set(key, next);
    return next;
  },
  on(fn) {
    listeners.add(fn);
    return () => listeners.delete(fn);
  },
  stamp(key) {
    return stamps[key] || 0;
  },
  /** Every synced key -> {v, t}. Keys starting with "_" stay on this device. */
  dump() {
    const out = {};
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (!k.startsWith(PREFIX)) continue;
      const key = k.slice(PREFIX.length);
      if (key.startsWith('_')) continue;
      out[key] = { v: localStorage.getItem(k), t: stamps[key] || 0 };
    }
    return out;
  },
  /** Take a remote dump; newer remote keys win. Returns the keys that changed. */
  merge(remote) {
    const changed = [];
    for (const [key, entry] of Object.entries(remote || {})) {
      if (!entry || typeof entry.v !== 'string' || key.startsWith('_')) continue;
      if ((entry.t || 0) > (stamps[key] || 0)) {
        localStorage.setItem(PREFIX + key, entry.v);
        stamps[key] = entry.t;
        changed.push(key);
      }
    }
    if (changed.length) {
      localStorage.setItem(PREFIX + '_t', JSON.stringify(stamps));
      listeners.forEach((fn) => fn('*', changed));
    }
    return changed;
  },
};

async function getJson(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`${url}: ${res.status}`);
  return res.json();
}

export async function loadData() {
  const [phrases, dialogs, guide, lessons, alphabet] = await Promise.all([
    getJson('../phrases.json'), getJson('../dialogs.json'), getJson('data/grammar_guide.json'),
    getJson('data/grammar_lessons.json'), getJson('data/alphabet.json'),
  ]);
  const topics = new Map();
  phrases.forEach((p, i) => {
    // Same key the Android app stores progress under.
    p.id = L.cardKey(p.coreIndex ?? i, p.category, p.polish);
    data.byId.set(p.id, p);
    const plain = L.normalizeHeadword(p.polish);
    if (!data.byPolish.has(plain)) data.byPolish.set(plain, p);
    const t = topics.get(p.category) || { name: p.category, count: 0, levels: {} };
    t.count++;
    t.levels[p.level] = (t.levels[p.level] || 0) + 1;
    topics.set(p.category, t);
  });
  Object.assign(data, { phrases, dialogs, guide, grammarLessons: lessons, alphabet, topics: [...topics.values()] });
  return data;
}

/** Leitner memory of one card: {box, dueAt}. box 0 = never graded. */
export function memOf(card) {
  const m = store.get('mem', {})[card.id];
  return m ? { box: m[0], dueAt: m[1] } : { box: 0, dueAt: 0 };
}

/** 0 = new, 1 = due for review, 2 = learnt (scheduled ahead). */
export function statusOf(card) {
  const m = memOf(card);
  return L.statusOf(m.box, m.dueAt, Date.now());
}

/** Record "Umiem!" (true) or "Jeszcze nie" (false) for a card. */
export function grade(card, got) {
  const now = Date.now();
  const box = L.nextBox(memOf(card).box, got);
  store.update('mem', (mem) => ({ ...mem, [card.id]: [box, L.nextDueAt(box, got, now)] }), {});
}

export function memoryCounts() {
  const counts = [0, 0, 0];
  const mem = store.get('mem', {});
  const now = Date.now();
  for (const p of data.phrases) {
    const m = mem[p.id];
    counts[m ? L.statusOf(m[0], m[1], now) : 0]++;
  }
  return counts; // [new, due, learnt]
}

  store.update('fav', (list) => (list.includes(id) ? list.filter((x) => x !== id) : [...list, id]), []);
  return isFav(id);
}

  return store.get('fav', []).map((f) => (typeof f === 'string' ? data.byId.get(f) : f)).filter(Boolean).reverse();
}

export function addToList(name, item) {
  store.update('lists', (all) => {
    const next = all.map((l) => ({ ...l, items: [...l.items] }));
    let list = next.find((l) => l.name === name);
    if (!list) next.push((list = { name, items: [] }));
    if (!list.items.some((x) => x.polish === item.polish)) list.items.push({ polish: item.polish, english: item.english || '' });
    return next;
  }, []);
}

export function noteFor(polish) {
  return store.get('notes', {})[L.normalizeHeadword(polish)] || '';

export function setNote(polish, text) {
  store.update('notes', (notes) => {
    const next = { ...notes };
    const key = L.normalizeHeadword(polish);
    if (text && text.trim()) next[key] = text.trim();
    else delete next[key];
    return next;
  }, {});
}

export const canSpeak = () => 'speechSynthesis' in window;
export const voicesFor = (lang) => voices.filter((v) => v.lang.toLowerCase().startsWith(lang.slice(0, 2).toLowerCase()));

let speakToken = 0;
/** Read text aloud. lang 'pl-PL' (default) or 'en-US'. Resolves when it ends, is cut off, or cannot play. */
export function speak(text, { lang = 'pl-PL', rate } = {}) {
  return new Promise((resolve) => {
    if (!canSpeak() || !text) return resolve(false);
    const token = ++speakToken;
    window.speechSynthesis.cancel();
    const u = new SpeechSynthesisUtterance(String(text).replace(/[_…]+/g, ' '));
    const s = settings();
    const wanted = lang.startsWith('pl') ? s.voicePl : s.voiceEn;
    const pool = voicesFor(lang);
    u.voice = pool.find((v) => v.voiceURI === wanted) || pool.find((v) => v.localService) || pool[0] || null;
    u.lang = lang;
    u.rate = rate || SPEEDS[s.speed] || 1;
    const done = (ok) => () => resolve(ok && token === speakToken);
    u.onend = done(true);
    u.onerror = done(false);
    window.speechSynthesis.speak(u);
  });
}

export function stopSpeaking() {
  speakToken++;
  if (canSpeak()) window.speechSynthesis.cancel();
}

const Recognition = window.SpeechRecognition || window.webkitSpeechRecognition;
export const canListen = () => !!Recognition;

/** Listen once. Resolves with what was heard (best guesses first, [] if nothing); rejects with Error(code) when the mic cannot be used. */
export function listen(lang = 'pl-PL') {
  return new Promise((resolve, reject) => {
    if (!Recognition) return reject(new Error('unsupported'));
    stopSpeaking();
    const rec = new Recognition();
    rec.lang = lang;
    rec.maxAlternatives = 5;
    rec.interimResults = false;
    let heard = [];
    rec.onresult = (e) => {
      heard = [...e.results[0]].map((alt) => alt.transcript);
    };
    rec.onerror = (e) => (e.error === 'no-speech' ? resolve([]) : reject(new Error(e.error)));
    rec.onend = () => resolve(heard);
    rec.start();
  });
}

export async function copyText(text) {
  try {
    await navigator.clipboard.writeText(text);
    return true;
  } catch (e) {
    const area = h('textarea', { style: { position: 'fixed', opacity: 0 } }, text);
    document.body.append(area);
    area.select();
    const ok = document.execCommand('copy');
    area.remove();
    return ok;
  }
}

export function downloadText(filename, text, type = 'text/markdown') {
  const url = URL.createObjectURL(new Blob([text], { type: `${type};charset=utf-8` }));
  h('a', { href: url, download: filename }).click();
  setTimeout(() => URL.revokeObjectURL(url), 2000);
}

- Spans: Polish spans -> ui.pl(t) (wrapped in <b> when b); other spans -> <b> or plain text.
- h -> h3. p -> <p>. list -> ul/ol. table -> h('div', {class: 'table-wrap'}, table with thead/tbody); honour `rows` by slicing.
- study -> a .chips wrap with one button (class 'chip') per entry (honour `words`): Polish text; clicking calls openWord({polish: entry}).
- dialogue -> ui.card with h3 title, each line as tappable(line, {tag: 'div'}), a small 'Play' button that speaks the lines in order, and the meaning in muted small text.
- exercise -> ui.card with the prompt as h3 and, per item, the question text, an <input class="input"> and a 'Check' button; on check (or Enter) show a verdict line under it using L.checkAnswer(typed, ...item.a, item.full): ANSWER_EXACT 'Correct', ANSWER_DIACRITICS 'Right, but watch the Polish letters: <full>', ANSWER_WORD_ORDER or ANSWER_CLOSE 'Almost: <full>', ANSWER_WRONG 'Not quite: <full>' (classes 'verdict ok' / 'verdict no'). This block manages its own small DOM updates (no ctx).
View:
- #/guide: ui.header(data.guide.title, {kicker: 'Reference'}), a short muted line 'The full guide, chapter by chapter. For step-by-step lessons with quizzes use the course.', a button to #/course, then the chapters as ui.row(`${id} · ${title}`, '→') in a .grid.grid-2 -> #/guide/<id>.
- #/guide/<id>: ui.header(chapter title, {back: '#/guide', kicker: `Chapter ${id}`}), all blocks inside h('article', {class: 'prose'}), then previous/next chapter buttons. Unknown id: back to the list.

--- course.js (#/course and #/course/<lessonId>) ---
Progress lives in store key 'course' = {done: {lessonId: bestPercent}, last: lessonId}.
- #/course (map): ui.header('Grammar course', {kicker: `${doneCount} of ${total} lessons done`}), ui.progress(doneCount / total), then one section per unit (unit.title) with ui.row(lesson.title, right, onclick, sub) per lesson: right = '✓ ' + best + '%' when done, else '→'; sub = `${quiz.length} questions` or 'Reading' when the quiz is empty. The first lesson not yet done gets a ui.button('Continue: <title>', {kind: 'primary', block: true}) at the top of the page.
- #/course/<lessonId>: find the lesson and its unit; chapter = data.guide.chapters[unit.chapter]. Phases in ctx.state.phase: 'learn' (default), 'quiz', 'done'. Store 'course'.last = lessonId when the lesson opens.
  - learn: ui.header(lesson.title, {back: '#/course', kicker: unit.title}); the referenced blocks via renderBlock(chapter.blocks[ref.b], {rows: ref.rows, words: ref.words}) inside article.prose; then ui.button(`Start the quiz · ${n} questions`, {kind: 'primary', block: true}), or 'Mark as done' when the quiz is empty (saves 100).
  - quiz: one question at a time (ctx.state.q index, ctx.state.score). Progress bar and `Question 2 of 6`. Prompt as h2.
    - mc: the options as ui.button(option, {block: true}) with class 'option' added; after a pick (ctx.state.picked) mark the correct one 'right' and a wrong pick 'wrong', disable them, show a verdict line ('Correct' / 'Not quite') and the explain text.
    - type: an input and a 'Check' button (Enter also checks); verdict from L.checkAnswer(typed, ...accepted) with the same wording as the guide exercise, always showing `model`. Count EXACT, DIACRITICS, WORD_ORDER and CLOSE as right.
    - After answering: if q.say is not empty a '🔊' button that speak()s it, and ui.button('Next', {kind: 'primary'}) ('Finish' on the last one).
  - done: percent = round(score / n * 100); save max(previous best, percent) into 'course'.done[lessonId]. Show kicker 'Lesson finished', h1 `${score} of ${n} correct`, and buttons 'Next lesson' (the following lesson in course order, if any), 'Try again', 'Back to the course'.
  Unknown lesson id: go back to #/course.
