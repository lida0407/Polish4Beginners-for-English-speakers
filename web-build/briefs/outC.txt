// Guide view: chapter browser and single-chapter reader.
import { h, ui, L, data, go, speak, stopSpeaking } from '../core.js';
import { openWord, tappable } from '../word.js';

// Render an individual span according to its styling flags.
function renderSpan(s) {
  if (!s) return null;
  if (s.pl) {
    const plNode = ui.pl(s.t);
    return s.b ? h('b', {}, plNode) : plNode;
  }
  return s.b ? h('b', {}, s.t) : s.t;
}

/** Render a single content block from the guide; reused by the course. */
export function renderBlock(block, { rows, words } = {}) {
  if (!block) return h('div');

  if (block.type === 'h') {
    return h('h3', {}, block.text || '');
  }

  if (block.type === 'p') {
    return h('p', {}, ...(block.spans || []).map(renderSpan));
  }

  if (block.type === 'list') {
    const tag = block.ordered ? 'ol' : 'ul';
    const items = (block.items || []).map((spans) => h('li', {}, ...spans.map(renderSpan)));
    return h(tag, {}, ...items);
  }

  if (block.type === 'table') {
    const allRows = block.rows || [];
    const tableRows = rows && Array.isArray(rows) && rows.length === 2
      ? allRows.slice(rows[0], rows[1])
      : allRows;
    const thead = h('thead', {}, h('tr', {}, ...(block.header || []).map((th) => h('th', {}, th))));
    const tbody = h('tbody', {}, ...tableRows.map((row) =>
      h('tr', {}, ...row.map((cell) => h('td', {}, ...(cell || []).map(renderSpan))))
    ));
    return h('div', { class: 'table-wrap' }, h('table', {}, thead, tbody));
  }

  if (block.type === 'study') {
    const allEntries = block.entries || [];
    const entries = Array.isArray(words)
      ? allEntries.filter((w) => words.includes(w))
      : allEntries;
    const chips = entries.map((entry) =>
      h('button', {
        class: 'chip',
        type: 'button',
        onclick: () => openWord({ polish: entry }),
      }, ui.pl(entry))
    );
    return h('div', { class: 'chips' }, ...chips);
  }

  if (block.type === 'dialogue') {
    const playBtn = ui.button('Play', {
      small: true,
      onclick: async () => {
        for (const line of block.lines || []) {
          const ok = await speak(line);
          if (!ok) break;
        }
      },
    });
    return ui.card({},
      h('div', { class: 'spread' }, h('h3', {}, block.title || ''), playBtn),
      h('div', { class: 'stack' }, ...(block.lines || []).map((line) => tappable(line, { tag: 'div' }))),
      block.meaning ? h('p', { class: 'muted small' }, block.meaning) : null
    );
  }

  if (block.type === 'exercise') {
    const items = (block.items || []).map((item) => {
      const input = h('input', { class: 'input', type: 'text', placeholder: 'Your answer...' });
      const verdict = h('div', { class: 'verdict' });
      const check = () => {
        const typed = input.value.trim();
        const res = L.checkAnswer(typed, ...(item.a || []), item.full);
        verdict.className = 'verdict ' + (res !== L.ANSWER_WRONG ? 'ok' : 'no');
        if (res === L.ANSWER_EXACT) {
          verdict.textContent = 'Correct';
        } else if (res === L.ANSWER_DIACRITICS) {
          verdict.textContent = `Right, but watch the Polish letters: ${item.full}`;
        } else if (res === L.ANSWER_WORD_ORDER || res === L.ANSWER_CLOSE) {
          verdict.textContent = `Almost: ${item.full}`;
        } else {
          verdict.textContent = `Not quite: ${item.full}`;
        }
      };
      input.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') check();
      });
      const checkBtn = ui.button('Check', { small: true, onclick: check });
      return h('div', { class: 'stack' },
        h('p', {}, item.q),
        h('div', { class: 'hstack' }, input, checkBtn),
        verdict
      );
    });
    return ui.card({},
      h('h3', {}, block.prompt || ''),
      h('div', { class: 'stack' }, ...items)
    );
  }

  return h('div');
}

    }
  }

  if (!ctx.params || ctx.params.length === 0) {

    const header = ui.header('Grammar course', { kicker: `${doneCount} of ${total} lessons done` });
    const progress = ui.progress(total > 0 ? doneCount / total : 0);

          block: true,
          onclick: () => go(`#/course/${encodeURIComponent(firstUndone.lesson.id)}`),

      : null;

      const rows = (u.lessons || []).map((l) => {
        const isDone = doneMap[l.id] !== undefined;
        const right = isDone ? `✓ ${doneMap[l.id]}%` : '→';
        const sub = (l.quiz && l.quiz.length > 0) ? `${l.quiz.length} questions` : 'Reading';
        return ui.row(l.title, right, () => go(`#/course/${encodeURIComponent(l.id)}`), sub);

      break;

        placeholder: 'Type your answer...',
        disabled: answered,

    return h('div', { class: 'stack' }, head, body, actionRow);

  const percent = n > 0 ? Math.round((score / n) * 100) : 100;

  if (!ctx.state.saved) {

      const current = c || { done: {}, last: lessonId };

      done[lessonId] = Math.max(prev, percent);
      return { ...current, done, last: lessonId };
    }, { done: {}, last: lessonId });

  const header = ui.header(n > 0 ? `${score} of ${n} correct` : 'Lesson completed', { kicker: 'Lesson finished' });

  const nextItem = currentIdx >= 0 && currentIdx < allLessons.length - 1 ? allLessons[currentIdx + 1] : null;

  const buttons = [
    nextItem
      ? ui.button('Next lesson', { kind: 'primary', block: true, onclick: () => go(`#/course/${encodeURIComponent(nextItem.lesson.id)}`) })
      : null,
    n > 0
      ? ui.button('Try again', {
          block: true,
          onclick: () => {
            ctx.state.phase = 'learn';
            ctx.state.score = 0;
            ctx.state.q = 0;
            delete ctx.state.saved;
            delete ctx.state.picked;
            delete ctx.state.checked;
            delete ctx.state.typed;
            delete ctx.state.answerRes;
            ctx.refresh();
          },
        })
      : null,
    ui.button('Back to the course', { block: true, onclick: () => go('#/course') }),
  ].filter(Boolean);

  return h('div', { class: 'stack' }, header, ...buttons);

// Alphabet view: letter tiles and pronunciation examples.
import { h, ui, data, speak, stopSpeaking } from '../core.js';

export default function view(ctx) {
  ctx.onLeave?.(() => stopSpeaking());
  const items = data.alphabet || [];
  const sel = typeof ctx.state.sel === 'number' ? ctx.state.sel : 0;
  const current = items[sel] || items[0] || { letter: '', sound: '', example: '', english: '' };

  const header = ui.header('Alphabet', { kicker: `${items.length} letters and sounds` });

  const hearBtn = ui.button('🔊 Hear the example', {
    kind: 'primary',
    onclick: () => speak(current.example),
  });

  const detailCard = ui.card({ class: 'card-soft' },
    h('div', { class: 'stack center' },
      h('div', { class: 'flash-front', style: { fontSize: '3rem' } }, current.letter),
      h('div', { class: 'muted' }, current.sound),
      h('div', {}, ui.pl(current.example), h('span', { class: 'muted' }, ` · ${current.english}`)),
      hearBtn
    )
  );

  const tiles = items.map((item, i) => {
    const selected = i === sel;
    return ui.card({
      class: 'center' + (selected ? ' card-hero' : ''),
      onclick: () => {
        ctx.state.sel = i;
        speak(item.example);
        ctx.refresh();
      },
    },
      h('h2', {}, item.letter),
      h('div', { class: 'small' }, item.sound)
    );
  });

  const grid = h('div', { class: 'grid grid-tiles' }, ...tiles);

  return h('div', { class: 'stack' }, header, detailCard, grid);
}
